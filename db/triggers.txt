/* update request timestamp when status changes */ 
DROP TRIGGER IF EXISTS trg_requests_updated_at;
DELIMITER $
CREATE TRIGGER trg_requests_updated_at
BEFORE UPDATE ON requests
FOR EACH ROW
BEGIN
	IF OLD.status <> NEW.status THEN
		SET NEW.updated_at = CURRENT_TIMESTAMP;
	END IF;
END$
DELIMITER ;


/* create delivery when request is accepted */
DROP TRIGGER IF EXISTS trg_create_delivery_after_accept;
DELIMITER $
CREATE TRIGGER trg_create_delivery_after_accept
AFTER UPDATE ON requests
FOR EACH ROW
BEGIN
	IF OLD.status <> 'ACCEPTED'
		AND NEW.status = 'ACCEPTED' THEN
		INSERT INTO deliveries(req_id,status)
		VALUES(NEW.rq_id,'PENDING');
	END IF;
END$
DELIMITER ;


/* decrease portions when request is accepted */
DROP TRIGGER IF EXISTS trg_reduce_portions_on_accept;
DELIMITER $
CREATE TRIGGER trg_reduce_portions_on_accept
AFTER UPDATE ON requests
FOR EACH ROW
BEGIN
	IF OLD.status <> 'ACCEPTED'
		AND NEW.status='ACCEPTED' THEN
		UPDATE listing
		SET portions = portions - 1
		WHERE lst_id = NEW.lst_id;
	END IF;
END$
DELIMITER ;

/* decrease credits when user's request is accepted */
DROP TRIGGER IF EXISTS trg_reduce_credits_on_accept;
DELIMITER $
CREATE TRIGGER trg_reduce_credits_on_accept
AFTER UPDATE ON requests
FOR EACH ROW
BEGIN
	IF OLD.status <> 'ACCEPTED'
		AND NEW.status='ACCEPTED' THEN
		UPDATE student
		SET credits = credits - 1
		WHERE student.std_id = NEW.std_id;
	END IF;
END$
DELIMITER ;

/* when listing portions become 0, change to FULL (as long as it is not expired) and when portions are added change to ACTIVE*/
DROP TRIGGER IF EXISTS trg_listing_full;
DELIMITER $
CREATE TRIGGER trg_listing_full
BEFORE UPDATE ON listing
FOR EACH ROW
BEGIN
    IF NEW.portions = 0 AND NEW.status <> 'EXPIRED' THEN
        SET NEW.status = 'FULL';
    
    ELSEIF NEW.portions > 0 AND OLD.status = 'FULL' AND NEW.status <> 'EXPIRED' AND NEW.expires_at > NOW() THEN
        SET NEW.status = 'ACTIVE';
    END IF;
END$
DELIMITER ;

/* once rated, delivery status becomes completed */
DROP TRIGGER IF EXISTS trg_rating_completes_delivery;
DELIMITER $
CREATE TRIGGER trg_rating_completes_delivery
BEFORE UPDATE ON deliveries
FOR EACH ROW
BEGIN
	IF NEW.rating IS NOT NULL AND OLD.rating IS NULL THEN
		SET NEW.status='COMPLETED';
		SET NEW.rating_timestamp=CURRENT_TIMESTAMP;
	END IF;
END$
DELIMITER ;


/* update listing average rating */
DROP TRIGGER IF EXISTS trg_update_listing_rating;
DELIMITER $
CREATE TRIGGER trg_update_listing_rating
AFTER UPDATE ON deliveries
FOR EACH ROW
BEGIN
	IF NEW.rating IS NOT NULL THEN
		UPDATE listing l
		SET lst_rating =
		(
			SELECT AVG(d.rating)
			FROM deliveries d
			JOIN requests r ON d.req_id=r.rq_id
			WHERE r.lst_id=l.lst_id
			AND d.rating IS NOT NULL
		)
		WHERE l.lst_id=
		(
			SELECT lst_id
			FROM requests
			WHERE rq_id=NEW.req_id
		);
	END IF;
END$
DELIMITER ;


/* update poster rating */
DROP TRIGGER IF EXISTS trg_update_student_rating;
DELIMITER $
CREATE TRIGGER trg_update_student_rating
AFTER UPDATE ON deliveries
FOR EACH ROW
BEGIN
	IF NEW.rating IS NOT NULL THEN
		UPDATE student s
		SET std_rating =
		(
			SELECT AVG(l.lst_rating)
			FROM listing l
			WHERE l.poster=s.std_id
			AND l.lst_rating IS NOT NULL
		)
		WHERE s.std_id=
		(
			SELECT l.poster
			FROM listing l
			JOIN requests r ON l.lst_id=r.lst_id
			WHERE r.rq_id=NEW.req_id
		);
	END IF;
END$
DELIMITER ;


/* add 48 hours active time to new listings */
DROP TRIGGER IF EXISTS set_expires_at;
DELIMITER $
CREATE TRIGGER set_expires_at
BEFORE INSERT ON listing
FOR EACH ROW
BEGIN
    IF NEW.expires_at IS NULL THEN
        SET NEW.expires_at = DATE_ADD(CURRENT_TIMESTAMP, INTERVAL 48 HOUR);
    END IF;
END$
DELIMITER ;

/*insert into admin/student after insert on user table*/

DROP TRIGGER IF EXISTS user_reg;
DELIMITER $$
CREATE TRIGGER user_reg
AFTER INSERT ON user
FOR EACH ROW
BEGIN
    IF NEW.usr_role ='student' THEN
        INSERT INTO student(std_id) VALUES (NEW.usr_id);
    ELSEIF NEW.usr_role = 'admin' THEN
        INSERT INTO admin(adm_id) VALUES (NEW.usr_id);
    END IF;
END $$

DELIMITER ;

/*update poster's given_meals upon delivery (only from pending to delivered)*/
DROP TRIGGER IF EXISTS increment_student_given_meals;
DELIMITER $$
CREATE TRIGGER increment_student_given_meals
AFTER UPDATE ON deliveries
FOR EACH ROW
BEGIN
	IF NEW.status = 'DELIVERED' AND OLD.status = 'PENDING' THEN
		UPDATE student SET given_meals = given_meals + 1 WHERE std_id = (
			SELECT listing.poster FROM listing
			JOIN requests ON requests.lst_id = listing.lst_id
			WHERE requests.rq_id = NEW.req_id
		);
	END IF;
END $$
DELIMITER ;

/*penalty for user who did not pick up or rate a delivery*/
DROP TRIGGER IF EXISTS credit_penalty;
DELIMITER $$
CREATE TRIGGER credit_penalty
AFTER UPDATE ON deliveries
FOR EACH ROW
BEGIN
	IF NEW.status = 'REJECTED' AND OLD.status IN ('PENDING', 'DELIVERED') THEN
		UPDATE student SET credits = credits - 1 WHERE std_id = (
			SELECT requests.std_id FROM requests
			WHERE requests.rq_id = NEW.req_id
		);
	END IF;
END $$
DELIMITER ;


DROP TRIGGER IF EXISTS trg_reject_req_del_on_listing_expiry;
DELIMITER $
CREATE TRIGGER trg_reject_req_del_on_listing_expiry
AFTER UPDATE ON listing
FOR EACH ROW
BEGIN
    IF (NEW.status='EXPIRED' AND OLD.status IN ('ACTIVE', 'FULL')) THEN
        UPDATE deliveries SET status='REJECTED' WHERE req_id IN (SELECT rq_id FROM requests WHERE lst_id=NEW.lst_id) AND status='PENDING';
        UPDATE requests SET status='REJECTED', updated_at=CURRENT_TIMESTAMP() WHERE lst_id=NEW.lst_id AND status='PENDING';
    END IF ;
END$
DELIMITER ;
